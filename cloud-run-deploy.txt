# ─── SECTION 9: Deploy to Cloud Run (production) ───

# Create Artifact Registry
gcloud artifacts repositories create semantic-layer `
    --repository-format=docker `
    --location=europe-west2

gcloud auth configure-docker europe-west2-docker.pkg.dev

# Build and push
docker build -t europe-west2-docker.pkg.dev/acn-uki-ds-data-ai-project/semantic-layer/api:latest .
docker push europe-west2-docker.pkg.dev/acn-uki-ds-data-ai-project/semantic-layer/api:latest

# Deploy
gcloud run deploy agentic-semantic-layer `
    --image=europe-west2-docker.pkg.dev/acn-uki-ds-data-ai-project/semantic-layer/api:latest `
    --region=europe-west2 `
    --platform=managed `
    --no-allow-unauthenticated `
    --set-env-vars="GOOGLE_GENAI_USE_VERTEXAI=TRUE,GOOGLE_CLOUD_PROJECT=acn-uki-ds-data-ai-project,GOOGLE_CLOUD_LOCATION=europe-west2,GCP_PROJECT=acn-uki-ds-data-ai-project,GCP_REGION=europe-west2,BQ_PROJECT=acn-uki-ds-data-ai-project,BQ_DATASET=customer_sales_data_ai_agentic_semantic_layer,NEO4J_URI=bolt://YOUR_NEO4J_HOST:7687,NEO4J_USER=neo4j,NEO4J_PASSWORD=your-password,VS_INDEX_ENDPOINT_ID=YOUR_ENDPOINT_ID,VS_TABLES_INDEX_ID=YOUR_ID,VS_KPIS_INDEX_ID=YOUR_ID,VS_CONCEPTS_INDEX_ID=YOUR_ID,VS_EXAMPLES_INDEX_ID=YOUR_ID,EMBEDDING_MODEL=text-embedding-005,GENERATIVE_MODEL=gemini-2.5-flash" `
    --memory=2Gi `
    --cpu=2 `
    --min-instances=1 `
    --max-instances=10 `
    --timeout=300


# ─── SECTION 10: Test deployed API ───

$URL = gcloud run services describe agentic-semantic-layer `
    --region=europe-west2 --format="value(status.url)"
$TOKEN = gcloud auth print-identity-token

# Health check
Invoke-RestMethod -Uri "$URL/health" `
    -Headers @{ Authorization = "Bearer $TOKEN" }

# Test diagnostic query
Invoke-RestMethod -Uri "$URL/query" `
    -Method POST -ContentType "application/json" `
    -Headers @{ Authorization = "Bearer $TOKEN" } `
