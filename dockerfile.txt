FROM python:3.12-slim

WORKDIR /app

# Install dependencies
COPY semantic_layer_agent/requirements.txt ./requirements.txt
RUN pip install --no-cache-dir -r requirements.txt

# Copy agent package
COPY semantic_layer_agent/ ./semantic_layer_agent/

# ADK web serves from the parent of the agent package
EXPOSE 8000

CMD ["adk", "api_server", "--agent_dir", "semantic_layer_agent", "--host", "0.0.0.0", "--port", "8000"]
